cmake_minimum_required(VERSION 3.22.1)
project(linkup_tox C)
set(TOX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../.native-deps/tox")
add_library(toxcore SHARED IMPORTED)
set_target_properties(toxcore PROPERTIES IMPORTED_LOCATION "${TOX_ROOT}/jni/${ANDROID_ABI}/libtoxcore.so")
add_library(linkup_tox SHARED linkup_tox.c)
target_include_directories(linkup_tox PRIVATE "${TOX_ROOT}/toxcore-android-arm64-v8a/include")
target_link_libraries(linkup_tox toxcore)
target_link_options(linkup_tox PRIVATE "-Wl,-z,max-page-size=16384")
